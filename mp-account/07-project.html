🔴 第 7 章 · 完整实战项目：签到通知 ​

本章信息

难度：🔴 精通 · 预计学习 90 分钟 · 前置：第 1–6 章（token、菜单、OAuth、消息、模板消息全部会用到）

🎯 学习目标 ​

读完本章，你将能够：

  • 能搭出 Express 后端的标准目录结构（routes / services 分层）
  • 能把 access_token 管理收敛成统一模块，全项目只认它
  • 能串起「菜单 → 消息回复 → 模板通知 → OAuth 网页」的完整闭环
  • 能用内网穿透把项目真正接入微信并跑通验收

📇 术语卡 ​

术语白话解释
ExpressNode.js 最流行的 Web 框架，几行代码就能起一个 HTTP 服务
路由（route）「哪个网址由哪段代码处理」的分发表
service 层放业务逻辑的目录，路由只管收发，脏活累活都交给它
事件推送用户点 click 菜单时，微信 POST 给你的那条 XML（MsgType 为 event）
环境变量存放密钥的「系统级盒子」，避免把 Secret 写死在代码里提交
完整闭环用户操作到系统反馈连成一圈：点菜单 → 签到 → 回复 → 通知 → 查进度

🚗 项目总览：「签到通知」要做什么 ​

一个完整闭环，把六章所学一次装车：

  1. 用户进入公众号，底部有菜单「今日签到」（click）和「我的主页」（view）
  2. 点「今日签到」→ 微信推送事件 → 后端记签到 → 被动回复「签到成功，连续 N 天」→ 同时发一条模板消息到「服务通知」
  3. 点「我的主页」→ OAuth 授权 → 网页展示用户昵称和连续签到天数

权限说明： 个人订阅号没有网页授权与模板消息权限，本项目用测试号即可全部跑通（第 1 章讲过沙盒页入口）；换成正式认证服务号时只需改配置，代码一行不动。

// services/token.js —— access_token 统一管理（第 2 章原版 + 模块化）
const config = require('../config')
let cache = { token: '', expireAt: 0 }
async function getToken() {
  if (cache.token && Date.now() < cache.expireAt) return cache.token
  const url = 'https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid=' + config.appid + '&secret=' + config.secret
  const data = await (await fetch(url)).json()
  if (!data.access_token) throw new Error('获取 access_token 失败：' + JSON.stringify(data))
  cache = { token: data.access_token, expireAt: Date.now() + (data.expires_in - 300) * 1000 }
  return cache.token
}
module.exports = { getToken }

// services/menu.js —— 运行 node services/menu.js 创建菜单
const config = require('../config')
const { getToken } = require('./token')
const menu = {

  const token = await getToken()
  const url = 'https://api.weixin.qq.com/cgi-bin/menu/create?access_token=' + token

    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(menu)

  console.log('菜单创建结果：', JSON.stringify(data))
}

📅 第 4 步：services/checkin.js（签到业务） ​

js
// services/checkin.js —— 简化版：数据存内存，重启清零；真实项目请换数据库
const records = new Map() // openid -> { lastDay: '2026-10-22', days: 3 }
function doCheckin(openid) {
  const today = new Date().toISOString().slice(0, 10)
  const rec = records.get(openid) || { lastDay: '', days: 0 }
  if (rec.lastDay === today) {
    return { ok: false, msg: '今天已经签到过啦', days: rec.days }
  }
  rec.lastDay = today
  rec.days += 1
  records.set(openid, rec)
  return { ok: true, msg: '签到成功', days: rec.days }
}
function getDays(openid) {
  const rec = records.get(openid)
  return rec ? rec.days : 0
}
module.exports = { doCheckin, getDays }

📢 第 5 步：services/notify.js（模板通知） ​

js
// services/notify.js —— 签到成功后推送模板消息（第 6 章）
const config = require('../config')
const { getToken } = require('./token')
async function sendCheckinNotify(openid, days) {
  const token = await getToken()
  const url = 'https://api.weixin.qq.com/cgi-bin/message/template/send?access_token=' + token
  const data = await (await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      touser: openid,
      template_id: config.templateId,
      data: {
        first: { value: '签到成功', color: '#173177' },
        keyword1: { value: new Date().toISOString().slice(0, 10) },
        keyword2: { value: '连续签到 ' + days + ' 天' },
        remark: { value: '打开「我的主页」查看进度', color: '#10B981' }
      }
    })
  })).json()
  if (data.errcode !== 0) {
    console.error('模板消息发送失败：', JSON.stringify(data)) // 通知失败不拖垮主流程
    return false
  }
  return true
}
module.exports = { sendCheckinNotify }

📞 第 6 步：routes/wechat.js（验证 + 回复 + 触发签到） ​

js
// routes/wechat.js —— echostr 验证 + 消息/事件处理（第 5 章 + 第 2 章 click 事件）
const express = require('express')
const crypto = require('crypto')
const config = require('../config')
const { doCheckin } = require('../services/checkin')
const { sendCheckinNotify } = require('../services/notify')
const router = express.Router()

function verify(signature, timestamp, nonce) {
  const str = [config.token, timestamp, nonce].sort().join('')
  return crypto.createHash('sha1').update(str).digest('hex') === signature
}
// 从 XML 里取某个标签的 CDATA 值（学习期够用，生产建议用成熟解析库）
function getTag(xml, tag) {
  const re = new RegExp('<' + tag + '><!\\[CDATA\\[(.*?)\\]\\]></' + tag + '>')
  const m = xml.match(re)
  return m ? m[1] : ''
}
function textReply(fromUser, toUser, content) {
  return '<xml>' +
    '<ToUserName><![CDATA[' + fromUser + ']]></ToUserName>' +
    '<FromUserName><![CDATA[' + toUser + ']]></FromUserName>' +
    '<CreateTime>' + Math.floor(Date.now() / 1000) + '</CreateTime>' +
    '<MsgType><![CDATA[text]]></MsgType>' +
    '<Content><![CDATA[' + content + ']]></Content>' +
    '</xml>'
}
function handleCheckin(openid) {
  const result = doCheckin(openid)
  if (result.ok) {
    // 不 await：确保 5 秒内先完成被动回复，通知异步补发
    sendCheckinNotify(openid, result.days).catch(() => {})
  }
  return result.msg + '！已连续 ' + result.days + ' 天'
}
router.get('/', (req, res) => {
  const q = req.query
  if (verify(q.signature, q.timestamp, q.nonce)) res.end(q.echostr) // 原样返回，一个字符都不能多
  else res.end('verify failed')
})
router.post('/', (req, res) => {
  let body = ''
  req.on('data', (chunk) => (body += chunk))
  req.on('end', () => {
    const user = getTag(body, 'FromUserName')
    const account = getTag(body, 'ToUserName')
    const msgType = getTag(body, 'MsgType')
    const content = getTag(body, 'Content')
    let replyText = '收到～发送「帮助」查看玩法'
    if (msgType === 'text' && content === '帮助') {
      replyText = '点菜单「今日签到」，或直接发送：签到'
    } else if (
      (msgType === 'text' && content === '签到') ||
      // click 事件：Event 为 CLICK，EventKey 就是创建菜单时的 key
      (msgType === 'event' && getTag(body, 'EventKey') === 'DO_CHECKIN')
    ) {
      replyText = handleCheckin(user)
    }
    res.writeHead(200, { 'Content-Type': 'text/xml; charset=utf-8' })
    // 注意：ToUserName / FromUserName 反转 —— 第一个是用户，第二个是公众号
    res.end(textReply(user, account, replyText))
  })
})

module.exports = router

🌐 第 7 步：routes/web.js（OAuth 主页） ​

js
// routes/web.js —— OAuth 登录 + 我的主页（第 3 章）
const express = require('express')
const config = require('../config')
const { getDays } = require('../services/checkin')

const router = express.Router()

router.get('/login', (req, res) => {
  const redirect = encodeURIComponent(config.baseUrl + '/oauth/callback')
  const authUrl =
    'https://open.weixin.qq.com/connect/oauth2/authorize?appid=' + config.appid +
    '&redirect_uri=' + redirect + '&response_type=code&scope=snsapi_userinfo&state=checkin#wechat_redirect'
  res.redirect(authUrl)
})

router.get('/oauth/callback', async (req, res) => {
  const code = req.query.code
  if (!code) { res.send('缺少 code，请从 /login 重新进入'); return }
  try {
    const tokenUrl =
      'https://api.weixin.qq.com/sns/oauth2/access_token?appid=' + config.appid +
      '&secret=' + config.secret + '&code=' + code + '&grant_type=authorization_code'
    const token = await (await fetch(tokenUrl)).json()
    if (!token.openid) throw new Error('换票失败：' + JSON.stringify(token))
    const userUrl = 'https://api.weixin.qq.com/sns/userinfo?access_token=' + token.access_token +
      '&openid=' + token.openid + '&lang=zh_CN'
    const user = await (await fetch(userUrl)).json()

    res.send('<h1>你好，' + user.nickname + '！</h1>' +
      '<p>连续签到：' + getDays(token.openid) + ' 天</p>' +
      '<img src="' + user.headimgurl + '" width="64"/>')
  } catch (e) {
    res.send('出错了：' + e.message)
  }
})

module.exports = router

💻 本地运行与内网穿透全流程 ​

从零运行步骤：

  1. 按「第 2 步」填好 config.js（appid、secret、templateId），测试号沙盒页新增第 6 章那套占位符的模板
  2. 启动内网穿透：cpolar http 3000（或 natapp 对应命令），复制公网域名填进 config.baseUrl（形如 https://xxxx.cpolar.io，结尾不带斜杠）
  3. 运行 node services/menu.js 创建菜单，再运行 node app.js 起服务
  4. 测试号沙盒页「接口配置信息」：URL 填 config.baseUrl + /wechat，Token 填 checkinBot123，点提交
  5. ✅ 你应该看到： 提交成功（echostr 校验通过）
  6. 手机关注测试号，点菜单「今日签到」
  7. ✅ 你应该看到： 会话里秒回「签到成功！已连续 1 天」，稍后「服务通知」弹出模板卡片「连续签到 1 天」；再点一次提示「今天已经签到过啦！已连续 1 天」
  8. 点菜单「我的主页」，同意授权
  9. ✅ 你应该看到： 网页显示「你好，你的昵称！」、头像和「连续签到：1 天」——菜单、事件、回复、通知、OAuth 全部闭环

✏️ 分级练习 ​

  1. 🟢 跟做型：完整跑通上面 9 步，把每一步「你应该看到」的结果截图存档，这就是你的项目验收单。
  2. 🟡 修改型：把 checkin.js 的内存 Map 改成 JSON 文件持久化（用 fs 读写），重启服务后签到天数不丢失。
  3. 🔴 创作型：加一个「排行榜」能力：新菜单项 + 网页路由 /rank，按签到天数倒序展示前 10 名（提示：Map 转数组再排序）。

⚠️ 常见坑 ​

  • 点「今日签到」毫无反应 → 服务器配置没启用，或 EventKey 与 menu.js 里的 DO_CHECKIN 大小写不一致 → 先确认后台配置处于「已启用」，再逐字符核对 key
  • 「我的主页」报 errcode: 40163, errmsg: "code been used" → config.baseUrl 与后台登记的网页授权域名不一致，用户被送到错误的回调页反复消费同一个 code → 统一改 config 后重启并重跑菜单脚本
  • 隧道重启后域名变了，菜单跳转 404 → view 菜单里的旧域名还写在线上 → 更新 config.baseUrl，重新运行 node services/menu.js 覆盖菜单
  • 回复内容偶发「该公众号暂时无法提供服务」 → 签到逻辑里同步等待了模板消息接口，5 秒内没完成被动回复 → 通知保持「不 await、只 catch」的异步姿势，被动回复永远最先返回
